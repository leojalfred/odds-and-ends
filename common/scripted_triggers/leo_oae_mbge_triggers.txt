### Leo VI's Odds and Ends - Mass Bolster Governance, eligibility.
###
### Who may be bolstered is vanilla's decision, not ours. The pool is whoever
### boost_efficiency_interaction would let the player target right now, and the
### mass action must never reach past that.

# Current scope: a candidate governor. Requires scope:leo_oae_mbge_actor, the
# player doing the bolstering.
#
# is_character_interaction_valid is the engine asking the interaction itself,
# so a patch that regates Bolster Governance regates this for free. It is
# evaluated on the actor with the target named as the recipient, which is why
# the governor has to be staged into a scope first.
#
# The two conditions after it are the interaction's own can_send block
# (06_ep3_interactions.txt:5711-5731), repeated because it is not certain the
# engine trigger covers can_send as well as is_shown and is_valid. They are
# cheap, and repeating a condition can only ever narrow the pool. Re-diff them
# against vanilla after a game patch.
#
# governor_efficiency_presented is staged rather than compared in place: a
# computed value on the left of a comparison is not reliable in every trigger
# context, and staging it costs nothing here because the caller needs the
# number anyway.
#
# The staged name is leo_oae_mbge_cand and not leo_oae_mbge_gov, which would
# read better, because leo_oae_mbge_gov is a *permanent* scope: the row's
# checkbox hands its character over under that name, so the scripted GUI
# declares it in saved_scopes and it outlives the click. A temporary scope
# cannot take a name a permanent one already holds, and the engine refuses the
# save rather than shadowing it, so reusing the name would leave every read
# here pointing at whichever row was ticked last instead of the governor being
# iterated. The two names must stay distinct.
leo_oae_mbge_target_trigger = {
	save_temporary_scope_as = leo_oae_mbge_cand

	scope:leo_oae_mbge_actor = {
		is_character_interaction_valid = {
			recipient = scope:leo_oae_mbge_cand
			interaction = boost_efficiency_interaction
		}
	}

	save_temporary_scope_value_as = {
		name = leo_oae_mbge_eff
		value = governor_efficiency_presented
	}
	scope:leo_oae_mbge_eff <= boost_efficiency_maximum_value

	NOR = {
		is_target_in_variable_list = {
			name = efficiency_boosters
			target = scope:leo_oae_mbge_actor
		}
		is_target_in_variable_list = {
			name = efficiency_damagers
			target = scope:leo_oae_mbge_actor
		}
	}
}

# Current scope: a candidate governor. Requires scope:leo_oae_mbge_actor.
#
# Vanilla allows it and the player has not set this one aside. Everything that
# counts, prices or spends asks this; only the list on screen asks the trigger
# above, because a governor who has been set aside still has to be shown, or
# there would be no way to put them back.
#
# Being set aside is a variable on the governor rather than a list on the
# player, so that the panel can read it per row with a plain variable read.
# There is no datafunction for list membership, and the alternatives were a
# scripted GUI per row per frame or splitting the pool into two lists, which
# made rows jump between them on every click. It also means the test lives in
# the same scope as the thing it tests, with no cross-scope read at all.
#
# The variable is read behind has_variable rather than on its own. The recount
# seeds it on every governor it walks, but this trigger is also asked of
# governors the recount has never reached: the apply loop iterates live, so a
# governor who became eligible since the last recount arrives here unseeded.
# Reading an unset variable is not merely untidy, it errors three times over
# and leaves the comparison's answer up to the engine. Absent means not set
# aside, which is the same thing a seeded zero means.
leo_oae_mbge_included_trigger = {
	leo_oae_mbge_target_trigger = yes
	trigger_if = {
		limit = { has_variable = leo_oae_mbge_excl }
		NOT = { var:leo_oae_mbge_excl = 1 }
	}
}

# Current scope: the player. Requires scope:leo_oae_mbge_eff, the efficiency of
# the governor being considered.
#
# Whether that governor falls under the threshold currently selected. The
# governor's efficiency is the governor's; the only thing the player holds is
# which threshold was picked, which is why this is asked here.
#
# Every comparison is a staged value against a literal, which is the shape the
# counting pass uses and the only shape proven to work. Two earlier versions
# silently matched everything: one read the threshold across scopes from inside
# an iterator's limit, the other compared two staged values against each other
# (scope:eff < scope:cut). Neither erred, they just passed, and the count came
# out right anyway because the counting pass never used them. Keep the literals.
leo_oae_mbge_row_shown_trigger = {
	trigger_if =      { limit = { var:leo_oae_mbge_thresh = 1 } scope:leo_oae_mbge_eff < -40 }
	trigger_else_if = { limit = { var:leo_oae_mbge_thresh = 2 } scope:leo_oae_mbge_eff < -30 }
	trigger_else_if = { limit = { var:leo_oae_mbge_thresh = 3 } scope:leo_oae_mbge_eff < -20 }
	trigger_else_if = { limit = { var:leo_oae_mbge_thresh = 4 } scope:leo_oae_mbge_eff < -10 }
	trigger_else_if = { limit = { var:leo_oae_mbge_thresh = 5 } scope:leo_oae_mbge_eff < 0 }
	trigger_else_if = { limit = { var:leo_oae_mbge_thresh = 6 } scope:leo_oae_mbge_eff < 10 }
	trigger_else_if = { limit = { var:leo_oae_mbge_thresh = 7 } scope:leo_oae_mbge_eff < 20 }
	trigger_else_if = { limit = { var:leo_oae_mbge_thresh = 8 } scope:leo_oae_mbge_eff < 30 }
	# 9 is every eligible governor, 10 is the budget walk. Neither filters by
	# efficiency.
	trigger_else = { always = yes }
}

# Current scope: the player. Requires scope:leo_oae_mbge_actor and, when the
# selected option is Grease Palms, scope:leo_oae_mbge_gold_cost.
#
# Can one more governor be paid for out of what is actually in the treasury
# right now? Every place that spends checks this first, so a stale count on
# screen can only ever bolster fewer governors than advertised, never overdraw.
leo_oae_mbge_can_pay_one_trigger = {
	influence >= leo_oae_mbge_unit_influence_value
	trigger_if = {
		limit = { var:leo_oae_mbge_opt = 2 }
		gold >= scope:leo_oae_mbge_gold_cost
	}
}

# Current scope: the player. Is there anything for the Bolster button to do?
leo_oae_mbge_has_targets_trigger = {
	has_variable = leo_oae_mbge_count
	var:leo_oae_mbge_count > 0
}

# Current scope: the player. The feature's own gate, mirroring the actor half
# of the interaction's is_shown (06_ep3_interactions.txt:5663-5671) plus the
# adulthood its is_valid requires. Administrative government carries its own
# DLC, so this gates on ownership without naming a DLC.
leo_oae_mbge_available_trigger = {
	is_ruler = yes
	is_adult = yes
	government_allows = administrative
}
