### Leo VI's Odds and Ends - Mass Bolster Governance.
###
### Applies vanilla's Bolster Governance across the governor pool in one go.
### Nothing here reaches past what the player could do by hand: the pool is
### whatever vanilla's own trigger allows (leo_oae_mbge_triggers.txt), the price
### is vanilla's own cost block, and the outcome is vanilla's own effect.
###
### The interaction is deliberately not fired. run_interaction takes no
### send_option, so firing it would leave every scope:*_boost flag unset, the
### switch in its on_accept would fall through to the fallback, and the player
### would get a guaranteed success at the cheapest price with no duel rolled.
### Paying the cost here and calling the same effects vanilla's on_accept calls
### is what keeps the odds and the bill honest.
###
### Threshold buckets are addressed by a suffix rather than a number, because a
### script variable name cannot be built at runtime. Codes 1-10 map to
### t_m40 t_m30 t_m20 t_m10 t_0 t_10 t_20 t_30 t_all t_bud, in that order, and
### are append-only: they live in saved variables, so renumbering would rewrite
### the settings of everyone already playing.

### The dropdown contents.
#
# The panel's dropdowns are datamodels over these lists, and each entry's flag
# name is its whole identity: t_all gives the label key leo_oae_mbge_ui_t_all
# and the scripted GUI leo_oae_mbge_set_t_all. Adding an option means adding a
# flag here, a loc key, and a scripted GUI with the matching name.
leo_oae_mbge_build_options_effect = {
	clear_variable_list = leo_oae_mbge_opts
	add_to_variable_list = { name = leo_oae_mbge_opts target = flag:opt_1 }
	add_to_variable_list = { name = leo_oae_mbge_opts target = flag:opt_2 }
	add_to_variable_list = { name = leo_oae_mbge_opts target = flag:opt_3 }
	add_to_variable_list = { name = leo_oae_mbge_opts target = flag:opt_4 }
	add_to_variable_list = { name = leo_oae_mbge_opts target = flag:opt_5 }

	# The two that need no thinking about come first: everyone the game allows,
	# then everyone the treasury allows. The graduated levels follow for when a
	# run wants aiming. Display order is not code order, and deliberately so:
	# the codes are append-only because they persist in saves.
	clear_variable_list = leo_oae_mbge_threshes
	add_to_variable_list = { name = leo_oae_mbge_threshes target = flag:t_all }
	add_to_variable_list = { name = leo_oae_mbge_threshes target = flag:t_bud }
	add_to_variable_list = { name = leo_oae_mbge_threshes target = flag:t_m40 }
	add_to_variable_list = { name = leo_oae_mbge_threshes target = flag:t_m30 }
	add_to_variable_list = { name = leo_oae_mbge_threshes target = flag:t_m20 }
	add_to_variable_list = { name = leo_oae_mbge_threshes target = flag:t_m10 }
	add_to_variable_list = { name = leo_oae_mbge_threshes target = flag:t_0 }
	add_to_variable_list = { name = leo_oae_mbge_threshes target = flag:t_10 }
	add_to_variable_list = { name = leo_oae_mbge_threshes target = flag:t_20 }
	add_to_variable_list = { name = leo_oae_mbge_threshes target = flag:t_30 }

	clear_variable_list = leo_oae_mbge_orders
	add_to_variable_list = { name = leo_oae_mbge_orders target = flag:order_1 }
	add_to_variable_list = { name = leo_oae_mbge_orders target = flag:order_2 }

	# Stamps what these lists were built from, so a save carrying an older set
	# is rebuilt rather than trusted. Bump it whenever the entries or their
	# order change, or a save keeps showing the old list.
	set_variable = { name = leo_oae_mbge_opts_version value = 1 }

	# The branch never runs. See the trigger for why it has to be named.
	if = { limit = { leo_oae_mbge_ack_trigger = yes } }
}

# Current scope: the player. Every variable the panel reads must exist before
# it is read, because an undefined variable has no defined reading in GUI.
#
# The default is the case the feature is for: the governors actually dragging
# the realm down, paid for with influence alone.
leo_oae_mbge_init_defaults_effect = {
	if = {
		limit = {
			OR = {
				NOT = { has_variable = leo_oae_mbge_opts_version }
				NOT = { var:leo_oae_mbge_opts_version = 1 }
			}
		}
		leo_oae_mbge_build_options_effect = yes
	}

	if = {
		limit = { NOT = { has_variable = leo_oae_mbge_open } }
		set_variable = { name = leo_oae_mbge_open value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = leo_oae_mbge_opt } }
		set_variable = { name = leo_oae_mbge_opt value = 1 }
	}
	# Every eligible governor, which the recount downgrades on the spot if it
	# turns out not to be payable.
	if = {
		limit = { NOT = { has_variable = leo_oae_mbge_thresh } }
		set_variable = { name = leo_oae_mbge_thresh value = 9 }
	}
	if = {
		limit = { NOT = { has_variable = leo_oae_mbge_order } }
		set_variable = { name = leo_oae_mbge_order value = 1 }
	}
	if = {
		limit = { NOT = { has_variable = leo_oae_mbge_count } }
		set_variable = { name = leo_oae_mbge_count value = 0 }
		set_variable = { name = leo_oae_mbge_infl value = 0 }
		set_variable = { name = leo_oae_mbge_gold value = 0 }
		set_variable = { name = leo_oae_mbge_afford_sel value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = leo_oae_mbge_excluded_count } }
		set_variable = { name = leo_oae_mbge_excluded_count value = 0 }
	}
}

### Counting and pricing.

# Current scope: the player.
leo_oae_mbge_zero_bucket_effect = {
	set_variable = { name = leo_oae_mbge_count_$B$ value = 0 }
	set_variable = { name = leo_oae_mbge_gold_$B$ value = 0 }
	set_variable = { name = leo_oae_mbge_afford_$B$ value = 0 }
}

# Current scope: the player. Requires scope:leo_oae_mbge_gold_cost.
leo_oae_mbge_accum_bucket_effect = {
	change_variable = { name = leo_oae_mbge_count_$B$ add = 1 }
	change_variable = { name = leo_oae_mbge_gold_$B$ add = scope:leo_oae_mbge_gold_cost }
}

# Current scope: the player. Fills every bucket in one pass over the pool.
#
# The iterator is top_liege's vassals, not the player's own, because vanilla's
# is_shown asks for recipient.top_liege = actor.top_liege
# (06_ep3_interactions.txt:5669). A player who is themselves a governor may
# bolster their peers, and looping over root's vassals would quietly take that
# away.
#
# The buckets are cumulative and the checks are plain ifs rather than a chain:
# a governor below -40 is also below -30, and belongs in every bucket above
# their own.
leo_oae_mbge_recount_effect = {
	save_scope_as = leo_oae_mbge_actor

	leo_oae_mbge_zero_bucket_effect = { B = t_m40 }
	leo_oae_mbge_zero_bucket_effect = { B = t_m30 }
	leo_oae_mbge_zero_bucket_effect = { B = t_m20 }
	leo_oae_mbge_zero_bucket_effect = { B = t_m10 }
	leo_oae_mbge_zero_bucket_effect = { B = t_0 }
	leo_oae_mbge_zero_bucket_effect = { B = t_10 }
	leo_oae_mbge_zero_bucket_effect = { B = t_20 }
	leo_oae_mbge_zero_bucket_effect = { B = t_30 }
	leo_oae_mbge_zero_bucket_effect = { B = t_all }
	leo_oae_mbge_zero_bucket_effect = { B = t_bud }

	top_liege = {
		every_vassal = {
			limit = { leo_oae_mbge_included_trigger = yes }

			save_temporary_scope_value_as = {
				name = leo_oae_mbge_gold_cost
				value = medium_gold_value
			}
			save_temporary_scope_value_as = {
				name = leo_oae_mbge_eff
				value = governor_efficiency_presented
			}

			scope:leo_oae_mbge_actor = {
				leo_oae_mbge_accum_bucket_effect = { B = t_all }
			}
			if = {
				limit = { scope:leo_oae_mbge_eff < -40 }
				scope:leo_oae_mbge_actor = { leo_oae_mbge_accum_bucket_effect = { B = t_m40 } }
			}
			if = {
				limit = { scope:leo_oae_mbge_eff < -30 }
				scope:leo_oae_mbge_actor = { leo_oae_mbge_accum_bucket_effect = { B = t_m30 } }
			}
			if = {
				limit = { scope:leo_oae_mbge_eff < -20 }
				scope:leo_oae_mbge_actor = { leo_oae_mbge_accum_bucket_effect = { B = t_m20 } }
			}
			if = {
				limit = { scope:leo_oae_mbge_eff < -10 }
				scope:leo_oae_mbge_actor = { leo_oae_mbge_accum_bucket_effect = { B = t_m10 } }
			}
			if = {
				limit = { scope:leo_oae_mbge_eff < 0 }
				scope:leo_oae_mbge_actor = { leo_oae_mbge_accum_bucket_effect = { B = t_0 } }
			}
			if = {
				limit = { scope:leo_oae_mbge_eff < 10 }
				scope:leo_oae_mbge_actor = { leo_oae_mbge_accum_bucket_effect = { B = t_10 } }
			}
			if = {
				limit = { scope:leo_oae_mbge_eff < 20 }
				scope:leo_oae_mbge_actor = { leo_oae_mbge_accum_bucket_effect = { B = t_20 } }
			}
			if = {
				limit = { scope:leo_oae_mbge_eff < 30 }
				scope:leo_oae_mbge_actor = { leo_oae_mbge_accum_bucket_effect = { B = t_30 } }
			}
		}
	}

	leo_oae_mbge_budget_pass_effect = yes
	leo_oae_mbge_afford_pass_effect = yes
	leo_oae_mbge_select_bucket_effect = yes
	leo_oae_mbge_fallback_effect = yes
	leo_oae_mbge_build_targets_effect = yes
}

# Current scope: the player. The efficiency a governor has to be under to count
# as inside the selected threshold.
#
# Codes 9 and 10 are every eligible governor and the budget walk, neither of
# which filters by efficiency, so they get a cutoff nothing can reach: vanilla
# already refuses anyone above 40.
leo_oae_mbge_set_cut_effect = {
	if =      { limit = { var:leo_oae_mbge_thresh = 1 } set_variable = { name = leo_oae_mbge_cut value = -40 } }
	else_if = { limit = { var:leo_oae_mbge_thresh = 2 } set_variable = { name = leo_oae_mbge_cut value = -30 } }
	else_if = { limit = { var:leo_oae_mbge_thresh = 3 } set_variable = { name = leo_oae_mbge_cut value = -20 } }
	else_if = { limit = { var:leo_oae_mbge_thresh = 4 } set_variable = { name = leo_oae_mbge_cut value = -10 } }
	else_if = { limit = { var:leo_oae_mbge_thresh = 5 } set_variable = { name = leo_oae_mbge_cut value = 0 } }
	else_if = { limit = { var:leo_oae_mbge_thresh = 6 } set_variable = { name = leo_oae_mbge_cut value = 10 } }
	else_if = { limit = { var:leo_oae_mbge_thresh = 7 } set_variable = { name = leo_oae_mbge_cut value = 20 } }
	else_if = { limit = { var:leo_oae_mbge_thresh = 8 } set_variable = { name = leo_oae_mbge_cut value = 30 } }
	else =    {                                        set_variable = { name = leo_oae_mbge_cut value = 1000 } }
}

### Who the run will actually visit.

# Current scope: the player. Rebuilds the list the panel shows, last, once the
# selection has settled.
#
# This is the one place that asks the plain eligibility trigger rather than the
# one that honors exclusions: a governor the player has set aside still has to
# appear, or there would be no way to put them back.
#
# It sorts them into two lists rather than one, because GUI has no way to ask
# whether a row is in a variable list. There is no datafunction for it, so the
# only way to tick a row from a single mixed list would be a scripted GUI
# evaluated per row per frame. Splitting the pool here instead means each row's
# tick is a literal on its template and nothing is evaluated at all while the
# panel sits open. It is the same move as the dropdowns: let script decide which
# list a thing belongs in, and let GUI pick the list by name.
#
# Worst governed first, so the list reads in the order the budget would spend
# in. Under the budget row there is no threshold to filter by, so it shows
# everyone eligible and the summary says how many of them the money reaches.
leo_oae_mbge_build_targets_effect = {
	save_scope_as = leo_oae_mbge_actor
	clear_variable_list = leo_oae_mbge_targets
	clear_variable_list = leo_oae_mbge_skipped
	set_variable = { name = leo_oae_mbge_excluded_count value = 0 }

	leo_oae_mbge_set_cut_effect = yes
	save_temporary_scope_value_as = {
		name = leo_oae_mbge_cut
		value = var:leo_oae_mbge_cut
	}

	top_liege = {
		ordered_vassal = {
			order_by = leo_oae_mbge_worst_first_value
			max = leo_oae_mbge_everyone_value
			check_range_bounds = no
			limit = { leo_oae_mbge_target_trigger = yes }

			# The threshold is applied here rather than in the limit above,
			# because comparing two staged values needs them staged first.
			save_temporary_scope_value_as = {
				name = leo_oae_mbge_eff
				value = governor_efficiency_presented
			}
			if = {
				limit = { scope:leo_oae_mbge_eff < scope:leo_oae_mbge_cut }

				save_temporary_scope_as = leo_oae_mbge_gov
				scope:leo_oae_mbge_actor = {
					if = {
						limit = {
							is_target_in_variable_list = {
								name = leo_oae_mbge_excluded
								target = scope:leo_oae_mbge_gov
							}
						}
						add_to_variable_list = {
							name = leo_oae_mbge_skipped
							target = scope:leo_oae_mbge_gov
						}
						change_variable = { name = leo_oae_mbge_excluded_count add = 1 }
					}
					else = {
						add_to_variable_list = {
							name = leo_oae_mbge_targets
							target = scope:leo_oae_mbge_gov
						}
					}
				}
			}
		}
	}
}

### Setting one aside.

# Current scope: the player. Requires scope:leo_oae_mbge_gov.
#
# Exclusions live on the player, so an heir starts with none: they are a working
# note about this run, not a standing opinion about the governor.
leo_oae_mbge_toggle_exclude_effect = {
	if = {
		limit = {
			is_target_in_variable_list = {
				name = leo_oae_mbge_excluded
				target = scope:leo_oae_mbge_gov
			}
		}
		remove_list_variable = {
			name = leo_oae_mbge_excluded
			target = scope:leo_oae_mbge_gov
		}
	}
	else = {
		add_to_variable_list = {
			name = leo_oae_mbge_excluded
			target = scope:leo_oae_mbge_gov
		}
	}

	# The count and the bill both move when one is set aside, so they are worked
	# out again rather than left to the next dropdown.
	leo_oae_mbge_recount_effect = yes
}

leo_oae_mbge_clear_exclusions_effect = {
	clear_variable_list = leo_oae_mbge_excluded
	leo_oae_mbge_recount_effect = yes
}

# Current scope: the player. Keeps the selection on something payable.
#
# Which levels are affordable depends on what is in the treasury right now, so
# the default cannot simply be written once at startup: a save opened with less
# influence than last time would come up on a level the Bolster button refuses.
# This runs after every recount and only ever moves a selection that has become
# unpayable, so a level the player chose deliberately is left alone as long as
# it still works.
#
# Everyone the game allows is the preferred landing place. Failing that, the
# budget row, which is payable by construction because it only ever takes what
# the treasury covers. If even that is empty there is nobody to bolster, and the
# button stays dark wherever the selection sits.
leo_oae_mbge_fallback_effect = {
	if = {
		limit = { var:leo_oae_mbge_afford_sel = 0 }

		if = {
			limit = { var:leo_oae_mbge_afford_t_all = 1 }
			set_variable = { name = leo_oae_mbge_thresh value = 9 }
		}
		else = {
			set_variable = { name = leo_oae_mbge_thresh value = 10 }
		}

		leo_oae_mbge_select_bucket_effect = yes
	}
}

# Current scope: a governor. Requires scope:leo_oae_mbge_actor.
# One step of the budget walk: take this governor if what has been committed so
# far plus their price still fits, and otherwise pass them over. Passing over
# rather than stopping matters under Grease Palms, where a governor too rich to
# afford should not block the cheaper ones behind them.
leo_oae_mbge_budget_try_effect = {
	save_temporary_scope_value_as = {
		name = leo_oae_mbge_gold_cost
		value = medium_gold_value
	}

	scope:leo_oae_mbge_actor = {
		if = {
			limit = {
				influence >= leo_oae_mbge_next_influence_value
				trigger_if = {
					limit = { var:leo_oae_mbge_opt = 2 }
					gold >= leo_oae_mbge_next_gold_value
				}
			}
			change_variable = { name = leo_oae_mbge_count_t_bud add = 1 }
			change_variable = { name = leo_oae_mbge_gold_t_bud add = scope:leo_oae_mbge_gold_cost }
			change_variable = { name = leo_oae_mbge_bud_infl add = leo_oae_mbge_unit_influence_value }
		}
	}
}

# Current scope: the player. Prices the "as many as I can afford" row by walking
# the pool in spending order and committing until the money runs out.
#
# The ordering only ever changes the answer under Grease Palms, since every
# other option charges the same for every governor. That is why the panel shows
# the Order dropdown for that option alone.
leo_oae_mbge_budget_pass_effect = {
	save_scope_as = leo_oae_mbge_actor
	set_variable = { name = leo_oae_mbge_bud_infl value = 0 }

	if = {
		limit = {
			var:leo_oae_mbge_opt = 2
			var:leo_oae_mbge_order = 2
		}
		top_liege = {
			ordered_vassal = {
				order_by = leo_oae_mbge_cheapest_first_value
				max = leo_oae_mbge_everyone_value
				check_range_bounds = no
				limit = { leo_oae_mbge_included_trigger = yes }
				leo_oae_mbge_budget_try_effect = yes
			}
		}
	}
	else = {
		top_liege = {
			ordered_vassal = {
				order_by = leo_oae_mbge_worst_first_value
				max = leo_oae_mbge_everyone_value
				check_range_bounds = no
				limit = { leo_oae_mbge_included_trigger = yes }
				leo_oae_mbge_budget_try_effect = yes
			}
		}
	}
}

# Current scope: the player. Whether this bucket can be paid for in full. The
# panel disables and grays a threshold row by reading the matching variable.
leo_oae_mbge_afford_bucket_effect = {
	save_temporary_scope_value_as = {
		name = leo_oae_mbge_b_count
		value = var:leo_oae_mbge_count_$B$
	}

	if = {
		limit = {
			var:leo_oae_mbge_count_$B$ > 0
			influence >= leo_oae_mbge_bucket_influence_value
			trigger_if = {
				limit = { var:leo_oae_mbge_opt = 2 }
				gold >= var:leo_oae_mbge_gold_$B$
			}
		}
		set_variable = { name = leo_oae_mbge_afford_$B$ value = 1 }
	}
	else = {
		set_variable = { name = leo_oae_mbge_afford_$B$ value = 0 }
	}
}

# Current scope: the player. The budget row is affordable by construction, so
# it only asks whether it found anyone at all.
leo_oae_mbge_afford_pass_effect = {
	leo_oae_mbge_afford_bucket_effect = { B = t_m40 }
	leo_oae_mbge_afford_bucket_effect = { B = t_m30 }
	leo_oae_mbge_afford_bucket_effect = { B = t_m20 }
	leo_oae_mbge_afford_bucket_effect = { B = t_m10 }
	leo_oae_mbge_afford_bucket_effect = { B = t_0 }
	leo_oae_mbge_afford_bucket_effect = { B = t_10 }
	leo_oae_mbge_afford_bucket_effect = { B = t_20 }
	leo_oae_mbge_afford_bucket_effect = { B = t_30 }
	leo_oae_mbge_afford_bucket_effect = { B = t_all }

	if = {
		limit = { var:leo_oae_mbge_count_t_bud > 0 }
		set_variable = { name = leo_oae_mbge_afford_t_bud value = 1 }
	}
	else = {
		set_variable = { name = leo_oae_mbge_afford_t_bud value = 0 }
	}
}

# Current scope: the player. Copies the selected bucket onto the three
# variables the summary line reads.
leo_oae_mbge_pick_bucket_effect = {
	set_variable = { name = leo_oae_mbge_count value = var:leo_oae_mbge_count_$B$ }

	# Every bucket carries a gold total whether or not gold is what is being
	# spent, because the pass that fills them does not know yet which method
	# will be selected. Only Grease Palms actually charges it, so the summary
	# must read zero for the other four rather than quoting a bill that will
	# never arrive.
	if = {
		limit = { var:leo_oae_mbge_opt = 2 }
		set_variable = { name = leo_oae_mbge_gold value = var:leo_oae_mbge_gold_$B$ }
	}
	else = {
		set_variable = { name = leo_oae_mbge_gold value = 0 }
	}

	save_temporary_scope_value_as = {
		name = leo_oae_mbge_b_count
		value = var:leo_oae_mbge_count_$B$
	}
	set_variable = { name = leo_oae_mbge_infl value = leo_oae_mbge_bucket_influence_value }

	# Kept alongside so the Bolster button can ask whether what is currently
	# selected is payable without walking all ten buckets again.
	set_variable = { name = leo_oae_mbge_afford_sel value = var:leo_oae_mbge_afford_$B$ }
}

# Current scope: the player. Written out per code because a variable name
# cannot be built from a value at runtime.
leo_oae_mbge_select_bucket_effect = {
	if =      { limit = { var:leo_oae_mbge_thresh = 1 }  leo_oae_mbge_pick_bucket_effect = { B = t_m40 } }
	else_if = { limit = { var:leo_oae_mbge_thresh = 2 }  leo_oae_mbge_pick_bucket_effect = { B = t_m30 } }
	else_if = { limit = { var:leo_oae_mbge_thresh = 3 }  leo_oae_mbge_pick_bucket_effect = { B = t_m20 } }
	else_if = { limit = { var:leo_oae_mbge_thresh = 4 }  leo_oae_mbge_pick_bucket_effect = { B = t_m10 } }
	else_if = { limit = { var:leo_oae_mbge_thresh = 5 }  leo_oae_mbge_pick_bucket_effect = { B = t_0 } }
	else_if = { limit = { var:leo_oae_mbge_thresh = 6 }  leo_oae_mbge_pick_bucket_effect = { B = t_10 } }
	else_if = { limit = { var:leo_oae_mbge_thresh = 7 }  leo_oae_mbge_pick_bucket_effect = { B = t_20 } }
	else_if = { limit = { var:leo_oae_mbge_thresh = 8 }  leo_oae_mbge_pick_bucket_effect = { B = t_30 } }
	else_if = { limit = { var:leo_oae_mbge_thresh = 10 } leo_oae_mbge_pick_bucket_effect = { B = t_bud } }
	else =    {                                          leo_oae_mbge_pick_bucket_effect = { B = t_all } }
}

### Applying.

# Current scope: a governor. Requires scope:leo_oae_mbge_actor.
#
# One bolster, matching boost_efficiency_interaction step for step: its cost
# block, then its on_send, then its on_accept. The two effects at the end are
# vanilla's own and read nothing but scope:actor and scope:recipient, which is
# why they can be called without the interaction around them.
leo_oae_mbge_bolster_one_effect = {
	save_scope_as = recipient
	save_temporary_scope_value_as = {
		name = leo_oae_mbge_gold_cost
		value = medium_gold_value
	}

	# cost (06_ep3_interactions.txt:5733-5757)
	scope:leo_oae_mbge_actor = {
		change_influence = { value = leo_oae_mbge_unit_influence_loss_value }
		if = {
			limit = { var:leo_oae_mbge_opt = 2 }
			remove_short_term_gold = leo_oae_mbge_gold_cost_value
		}
	}

	# on_send (:5783-5791)
	add_to_variable_list = {
		name = efficiency_boosters
		target = scope:leo_oae_mbge_actor
		years = 2
	}

	# on_accept (:5795-5809). The three skill options roll vanilla's duel, which
	# can fail and still charges; the other two always land.
	if = {
		limit = { scope:leo_oae_mbge_actor = { var:leo_oae_mbge_opt = 3 } }
		boost_governor_efficiency_duel_effect = { SKILL = diplomacy }
	}
	else_if = {
		limit = { scope:leo_oae_mbge_actor = { var:leo_oae_mbge_opt = 4 } }
		boost_governor_efficiency_duel_effect = { SKILL = stewardship }
	}
	else_if = {
		limit = { scope:leo_oae_mbge_actor = { var:leo_oae_mbge_opt = 5 } }
		boost_governor_efficiency_duel_effect = { SKILL = intrigue }
	}
	else = {
		boost_governor_efficiency_success_effect = yes
	}
}

# Current scope: a governor. Requires scope:leo_oae_mbge_actor.
# Bolsters this governor if there is still enough in the treasury for them.
# Every path spends through this, so the count on screen going stale can only
# ever mean fewer bolsters than advertised, never an overdraft.
leo_oae_mbge_try_bolster_effect = {
	save_temporary_scope_value_as = {
		name = leo_oae_mbge_gold_cost
		value = medium_gold_value
	}

	if = {
		limit = {
			scope:leo_oae_mbge_actor = { leo_oae_mbge_can_pay_one_trigger = yes }
		}
		leo_oae_mbge_bolster_one_effect = yes
	}
}

# Current scope: the player. What the Bolster button runs.
#
# The budget row walks the pool in spending order and takes whoever still fits;
# every other row takes everyone under the chosen threshold. Both go through the
# same per-governor affordability guard, so they differ only in who they visit
# and in what order.
leo_oae_mbge_apply_effect = {
	save_scope_as = leo_oae_mbge_actor
	save_scope_as = actor

	if = {
		limit = { var:leo_oae_mbge_thresh = 10 }
		if = {
			limit = { var:leo_oae_mbge_opt = 2  var:leo_oae_mbge_order = 2 }
			top_liege = {
				ordered_vassal = {
					order_by = leo_oae_mbge_cheapest_first_value
					max = leo_oae_mbge_everyone_value
					check_range_bounds = no
					limit = { leo_oae_mbge_included_trigger = yes }
					leo_oae_mbge_try_bolster_effect = yes
				}
			}
		}
		else = {
			top_liege = {
				ordered_vassal = {
					order_by = leo_oae_mbge_worst_first_value
					max = leo_oae_mbge_everyone_value
					check_range_bounds = no
					limit = { leo_oae_mbge_included_trigger = yes }
					leo_oae_mbge_try_bolster_effect = yes
				}
			}
		}
	}
	else = {
		# Worst first here too, so a budget that runs short mid-run spends what
		# is left on the governors who needed it most.
		leo_oae_mbge_set_cut_effect = yes
		save_temporary_scope_value_as = {
			name = leo_oae_mbge_cut
			value = var:leo_oae_mbge_cut
		}

		top_liege = {
			ordered_vassal = {
				order_by = leo_oae_mbge_worst_first_value
				max = leo_oae_mbge_everyone_value
				check_range_bounds = no
				limit = { leo_oae_mbge_included_trigger = yes }
				leo_oae_mbge_try_bolster_in_bucket_effect = yes
			}
		}
	}

	leo_oae_mbge_recount_effect = yes
}

# Current scope: a governor. Requires scope:leo_oae_mbge_actor and
# scope:leo_oae_mbge_cut.
#
# The threshold is applied here rather than in the iterator's limit, because
# comparing the governor's efficiency against the cutoff needs both staged as
# values first, and staging is an effect.
leo_oae_mbge_try_bolster_in_bucket_effect = {
	save_temporary_scope_value_as = {
		name = leo_oae_mbge_eff
		value = governor_efficiency_presented
	}

	if = {
		limit = { scope:leo_oae_mbge_eff < scope:leo_oae_mbge_cut }
		leo_oae_mbge_try_bolster_effect = yes
	}
}
